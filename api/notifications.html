import { createClient } from "@libsql/client"; import { Resend } from "resend"; const db = createClient({ url: process.env.TURSO_DATABASE_URL, authToken: process.env.TURSO_AUTH_TOKEN, }); const resend = new Resend(process.env.RESEND_API_KEY); export default async function handler(req, res) { const { action } = req.query; // 1. POST: Send Notification & Email Broadcast if (req.method === "POST" && action === "send-notification") { try { const { recipientType, recipientEmail, title, message } = req.body; if (!title || !message) { return res.status(400).json({ success: false, error: "Title and message are required." }); } const timestamp = Date.now(); // Store in DB for student notifications.html display await db.execute({ sql: `INSERT INTO notifications (recipient_type, recipient_email, title, message, created_at) VALUES (?, ?, ?, ?, ?)`, args: [recipientType, recipientType === "single" ? recipientEmail : null, title, message, timestamp] }); // Handle Email Dispatch if (recipientType === "single") { await resend.emails.send({ from: "Nexus Pro ", to: [recipientEmail], subject: title, text: message }); } else { // Broadcast to all users in user database const usersRes = await db.execute("SELECT email FROM users WHERE role = 'student'"); const emails = usersRes.rows.map(u => u.email).filter(Boolean); if (emails.length > 0) { await resend.emails.send({ from: "Nexus Pro ", bcc: emails, // Use BCC for mass broadcast subject: title, text: message }); } } return res.status(200).json({ success: true }); } catch (err) { return res.status(500).json({ success: false, error: err.message }); } } // 2. GET: Fetch Notifications for Student View if (req.method === "GET" && action === "get-notifications") { try { const { email } = req.query; // Returns announcements sent to 'all' OR sent specifically to this student's email const result = await db.execute({ sql: `SELECT * FROM notifications WHERE recipient_type = 'all' OR recipient_email = ? ORDER BY id DESC`, args: [email || ""] }); return res.status(200).json({ success: true, notifications: result.rows }); } catch (err) { return res.status(500).json({ success: false, error: err.message }); } } return res.status(400).json({ error: "Invalid action" }); }
